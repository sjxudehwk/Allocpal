import React, { useState, useEffect } from 'react';
import { base44 } from '@/api/base44Client';
import { Button } from '@/components/ui/button';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import { 
  ArrowLeft, Calendar, Users, Copy, CheckCircle, 
  Clock, Loader2, Sparkles, AlertTriangle, RefreshCw,
  BarChart3, User, Edit, Trash2, UserCog, LogOut
} from 'lucide-react';
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuTrigger,
  DropdownMenuSeparator,
} from '@/components/ui/dropdown-menu';
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
} from "@/components/ui/alert-dialog";
import { format, differenceInDays } from 'date-fns';
import { createPageUrl } from '@/utils';
import { Link } from 'react-router-dom';
import TaskCard from '@/components/project/TaskCard';
import ContributionPanel from '@/components/project/ContributionPanel';
import EditProjectModal from '@/components/project/EditProjectModal';
import EditMembersModal from '@/components/project/EditMembersModal';
import TaskSwapModal from '@/components/project/TaskSwapModal';
import ManageTasksModal from '@/components/project/ManageTasksModal';
import FairnessModal from '@/components/project/FairnessModal';
import CompletionSummaryModal from '@/components/project/CompletionSummaryModal';
import EditMyNameModal from '@/components/project/EditMyNameModal';

export default function Project() {
  const [project, setProject] = useState(null);
  const [members, setMembers] = useState([]);
  const [tasks, setTasks] = useState([]);
  const [currentUser, setCurrentUser] = useState(null);
  const [isLoading, setIsLoading] = useState(true);
  const [copied, setCopied] = useState(false);
  const [isGeneratingTasks, setIsGeneratingTasks] = useState(false);
  const [showContribution, setShowContribution] = useState(false);
  const [showEditProject, setShowEditProject] = useState(false);
  const [showEditMembers, setShowEditMembers] = useState(false);
  const [showTaskSwap, setShowTaskSwap] = useState(false);
  const [selectedTaskForSwap, setSelectedTaskForSwap] = useState(null);
  const [showDeleteDialog, setShowDeleteDialog] = useState(false);
  const [isDeleting, setIsDeleting] = useState(false);
  const [isOwner, setIsOwner] = useState(false);
  const [showManageTasks, setShowManageTasks] = useState(false);
  const [showFairnessModal, setShowFairnessModal] = useState(false);
  const [showCompletionSummary, setShowCompletionSummary] = useState(false);
  const [showLeaveDialog, setShowLeaveDialog] = useState(false);
  const [isLeaving, setIsLeaving] = useState(false);
  const [showEditMyName, setShowEditMyName] = useState(false);
  const [myMember, setMyMember] = useState(null);

  useEffect(() => {
    loadProject();
  }, []);

  const loadProject = async () => {
    try {
      const urlParams = new URLSearchParams(window.location.search);
      const projectId = urlParams.get('id');
      
      if (!projectId) {
        window.location.href = createPageUrl('Dashboard');
        return;
      }

      const user = await base44.auth.me();
      setCurrentUser(user);

      const projects = await base44.entities.Project.filter({ id: projectId });
      if (projects.length === 0) {
        window.location.href = createPageUrl('Dashboard');
        return;
      }

      setProject(projects[0]);

      const projectMembers = await base44.entities.ProjectMember.filter({ project_id: projectId });
      setMembers(projectMembers);

      const projectTasks = await base44.entities.Task.filter({ project_id: projectId });
      setTasks(projectTasks);

      // Check if current user is owner
      const ownerMember = projectMembers.find(m => m.is_owner);
      setIsOwner(ownerMember && ownerMember.email === user.email);

    } catch (error) {
      console.error('Failed to load project:', error);
    }
    setIsLoading(false);
  };

  const copyJoinCode = () => {
    navigator.clipboard.writeText(project.join_code);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  const updateTaskStatus = async (taskId, newStatus) => {
    try {
      await base44.entities.Task.update(taskId, { status: newStatus });
      const updatedTasks = tasks.map(t => t.id === taskId ? { ...t, status: newStatus } : t);
      setTasks(updatedTasks);
      
      // Check if all tasks are completed
      if (newStatus === 'completed') {
        const allCompleted = updatedTasks.every(t => t.status === 'completed');
        if (allCompleted) {
          setShowCompletionSummary(true);
        }
      }
    } catch (error) {
      console.error('Failed to update task:', error);
    }
  };

  const handleProjectUpdated = (updatedProject) => {
    setProject(updatedProject);
    setShowEditProject(false);
  };

  const handleMembersUpdated = () => {
    setShowEditMembers(false);
    loadProject();
  };

  const handleTaskSwapped = () => {
    setShowTaskSwap(false);
    setSelectedTaskForSwap(null);
    loadProject();
  };

  const handleSwapRequest = (task) => {
    setSelectedTaskForSwap(task);
    setShowTaskSwap(true);
  };

  const handleDeleteProject = async () => {
    setIsDeleting(true);
    try {
      // Delete all tasks
      for (const task of tasks) {
        await base44.entities.Task.delete(task.id);
      }
      
      // Delete all members
      for (const member of members) {
        await base44.entities.ProjectMember.delete(member.id);
      }
      
      // Delete project
      await base44.entities.Project.delete(project.id);
      
      // Redirect to dashboard
      window.location.href = createPageUrl('Dashboard');
    } catch (error) {
      console.error('Failed to delete project:', error);
    }
    setIsDeleting(false);
  };

  const handleLeaveProject = async () => {
    setIsLeaving(true);
    try {
      // Find current user's member record
      const userMember = members.find(m => m.email === currentUser.email);
      if (!userMember) return;

      // Delete user's tasks
      const userTasks = tasks.filter(t => t.assigned_to === currentUser.email);
      for (const task of userTasks) {
        await base44.entities.Task.delete(task.id);
      }

      // Delete membership
      await base44.entities.ProjectMember.delete(userMember.id);

      // Redirect to dashboard
      window.location.href = createPageUrl('Dashboard');
    } catch (error) {
      console.error('Failed to leave project:', error);
    }
    setIsLeaving(false);
  };

  const generateTasks = async () => {
    setIsGeneratingTasks(true);
    try {
      const prompt = `You are a project management AI. Break down this project into specific, actionable tasks and assign them fairly to team members.

Project Title: ${project.title}
Project Description: ${project.description || 'No description provided'}
Deadline: ${project.deadline}

Team Members:
${members.map((m, i) => `${i + 1}. ${m.name} (${m.email})${m.skill ? ` - Skill: ${m.skill}` : ''}`).join('\n')}

Requirements:
- Create 4-8 specific tasks based on the project scope
- Each team member must have at least 1 task
- Consider member skills when assigning tasks
- Tasks should be achievable before the deadline
- Include a mix of priorities (low, medium, high)

Return a JSON array of tasks.`;

      const response = await base44.integrations.Core.InvokeLLM({
        prompt,
        response_json_schema: {
          type: 'object',
          properties: {
            tasks: {
              type: 'array',
              items: {
                type: 'object',
                properties: {
                  title: { type: 'string' },
                  description: { type: 'string' },
                  assigned_email: { type: 'string' },
                  assigned_name: { type: 'string' },
                  priority: { type: 'string', enum: ['low', 'medium', 'high'] }
                }
              }
            }
          }
        }
      });

      if (response.tasks && response.tasks.length > 0) {
        const taskPromises = response.tasks.map(task =>
          base44.entities.Task.create({
            project_id: project.id,
            title: task.title,
            description: task.description || '',
            assigned_to: task.assigned_email,
            assigned_name: task.assigned_name,
            status: 'not_started',
            priority: task.priority || 'medium',
          })
        );
        const newTasks = await Promise.all(taskPromises);
        setTasks([...tasks, ...newTasks]);
        await base44.entities.Project.update(project.id, { tasks_generated: true });
        setProject({ ...project, tasks_generated: true });
      }
    } catch (error) {
      console.error('Failed to generate tasks:', error);
    }
    setIsGeneratingTasks(false);
  };

  if (isLoading) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-slate-50">
        <div className="text-center">
          <Loader2 className="w-8 h-8 animate-spin text-indigo-600 mx-auto mb-4" />
          <p className="text-slate-600">Loading project...</p>
        </div>
      </div>
    );
  }

  if (!project) return null;

  const daysLeft = differenceInDays(new Date(project.deadline), new Date());
  const completedTasks = tasks.filter(t => t.status === 'completed').length;
  const progressPercent = tasks.length > 0 ? (completedTasks / tasks.length) * 100 : 0;

  return (
    <div className="min-h-screen bg-slate-50">
      {/* Header */}
      <header className="bg-white border-b sticky top-0 z-10">
        <div className="max-w-6xl mx-auto px-4 py-4 flex justify-between items-center">
          <div className="flex items-center gap-4">
            <Link to={createPageUrl('Dashboard')}>
              <Button variant="ghost" size="icon">
                <ArrowLeft className="w-5 h-5" />
              </Button>
            </Link>
            <div className="flex items-center gap-2">
              <img 
                src="https://qtrypzzcjebvfcihiynt.supabase.co/storage/v1/object/public/base44-prod/public/6951500dd50ea78b48c99ef1/cdfa39cae_63bdf392-8a47-4fd5-8336-0b9d3d18433b.jpg"
                alt="AllocPal Logo"
                className="w-9 h-9 rounded-lg object-cover"
              />
              <span className="text-xl font-bold text-slate-900">AllocPal</span>
            </div>
          </div>
          <div className="flex gap-2">
            <Button
              variant="outline"
              onClick={() => setShowFairnessModal(true)}
            >
              <BarChart3 className="w-4 h-4 mr-2" />
              Fairness
            </Button>
            {!isOwner && (
              <Button
                variant="outline"
                onClick={() => setShowLeaveDialog(true)}
                className="text-red-600 hover:text-red-700"
              >
                <LogOut className="w-4 h-4 mr-2" />
                Leave
              </Button>
            )}
            {isOwner && (
              <DropdownMenu>
                <DropdownMenuTrigger asChild>
                  <Button variant="outline">
                    <Edit className="w-4 h-4 mr-2" />
                    Manage
                  </Button>
                </DropdownMenuTrigger>
                <DropdownMenuContent align="end">
                  <DropdownMenuItem onClick={() => setShowEditProject(true)}>
                    <Edit className="w-4 h-4 mr-2" />
                    Edit Project
                  </DropdownMenuItem>
                  <DropdownMenuItem onClick={() => setShowEditMembers(true)}>
                      <UserCog className="w-4 h-4 mr-2" />
                      Manage Members
                    </DropdownMenuItem>
                    <DropdownMenuItem onClick={() => setShowManageTasks(true)}>
                      <CheckCircle className="w-4 h-4 mr-2" />
                      Manage Tasks
                    </DropdownMenuItem>
                  <DropdownMenuSeparator />
                  <DropdownMenuItem 
                    onClick={() => setShowDeleteDialog(true)}
                    className="text-red-600 focus:text-red-600"
                  >
                    <Trash2 className="w-4 h-4 mr-2" />
                    Delete Project
                  </DropdownMenuItem>
                </DropdownMenuContent>
              </DropdownMenu>
            )}
          </div>
        </div>
      </header>

      <main className="max-w-6xl mx-auto px-4 py-8">
        {/* Project Header */}
        <div className="bg-white rounded-2xl p-6 mb-6 shadow-sm">
          <div className="flex flex-col md:flex-row md:items-start md:justify-between gap-4 mb-6">
            <div>
              <h1 className="text-2xl md:text-3xl font-bold text-slate-900 mb-2">
                {project.title}
              </h1>
              {project.description && (
                <p className="text-slate-600">{project.description}</p>
              )}
            </div>
            <div className="flex flex-wrap gap-3">
              <Badge 
                variant="outline" 
                className={`flex items-center gap-1.5 px-3 py-1.5 ${
                  daysLeft < 0 ? 'border-red-200 text-red-700 bg-red-50' :
                  daysLeft <= 3 ? 'border-orange-200 text-orange-700 bg-orange-50' :
                  'border-slate-200'
                }`}
              >
                <Calendar className="w-4 h-4" />
                {format(new Date(project.deadline), 'MMM d, yyyy')}
                {daysLeft >= 0 && ` (${daysLeft} days left)`}
              </Badge>
              <Badge variant="outline" className="flex items-center gap-1.5 px-3 py-1.5">
                <Users className="w-4 h-4" />
                {members.length} members
              </Badge>
            </div>
          </div>

          {/* Progress Bar */}
          <div className="mb-6">
            <div className="flex justify-between text-sm text-slate-600 mb-2">
              <span>Project Progress</span>
              <span className="font-medium">{completedTasks}/{tasks.length} tasks completed</span>
            </div>
            <div className="h-3 bg-slate-100 rounded-full overflow-hidden">
              <div 
                className="h-full bg-slate-900 rounded-full transition-all duration-500"
                style={{ width: `${progressPercent}%` }}
              />
            </div>
          </div>

          {/* Join Code & File */}
          <div className="space-y-3">
            <div className="flex items-center justify-between bg-slate-50 rounded-xl p-4">
              <div>
                <p className="text-sm text-slate-700 font-medium mb-1">Share this code with your team</p>
                <div className="text-3xl font-mono font-bold text-slate-900 tracking-widest">
                  {project.join_code}
                </div>
              </div>
              <Button onClick={copyJoinCode} variant="outline" className="border-slate-300">
                {copied ? <CheckCircle className="w-4 h-4 mr-2 text-emerald-500" /> : <Copy className="w-4 h-4 mr-2" />}
                {copied ? 'Copied!' : 'Copy'}
              </Button>
            </div>
            
            {project.file_url && (
              <div className="bg-slate-50 rounded-xl p-4">
                <p className="text-sm text-slate-700 font-medium mb-2">Attached File</p>
                <a 
                  href={project.file_url} 
                  target="_blank" 
                  rel="noopener noreferrer"
                  className="text-sm text-blue-600 hover:text-blue-700 hover:underline"
                >
                  View Project File →
                </a>
              </div>
            )}
          </div>
        </div>

        <div className="grid lg:grid-cols-3 gap-6">
          {/* Main Content */}
          <div className="lg:col-span-3 space-y-6">
            {/* Tasks Warning */}
            {!project.tasks_generated && tasks.length === 0 && (
              <Card className="border-amber-200 bg-amber-50">
                <CardContent className="p-4">
                  <div className="flex items-start gap-3">
                    <AlertTriangle className="w-5 h-5 text-amber-600 mt-0.5" />
                    <div className="flex-1">
                      <h3 className="font-medium text-amber-800 mb-1">Tasks not generated yet</h3>
                      <p className="text-sm text-amber-700 mb-3">
                        AI couldn't generate tasks during project creation. Click below to try again.
                      </p>
                      <Button 
                        onClick={generateTasks} 
                        disabled={isGeneratingTasks}
                        size="sm"
                        className="bg-amber-600 hover:bg-amber-700"
                      >
                        {isGeneratingTasks ? (
                          <>
                            <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                            Generating...
                          </>
                        ) : (
                          <>
                            <RefreshCw className="w-4 h-4 mr-2" />
                            Generate Tasks
                          </>
                        )}
                      </Button>
                    </div>
                  </div>
                </CardContent>
              </Card>
            )}

            {/* Tasks List */}
            <Card>
              <CardHeader>
                <CardTitle className="flex items-center gap-2">
                  <CheckCircle className="w-5 h-5 text-slate-900" />
                  Tasks
                </CardTitle>
              </CardHeader>
              <CardContent>
                {tasks.length === 0 ? (
                  <div className="text-center py-8 text-slate-500">
                    <Clock className="w-12 h-12 mx-auto mb-3 text-slate-300" />
                    <p>No tasks yet. {!project.tasks_generated && 'Generate tasks using the button above.'}</p>
                  </div>
                ) : (
                  <div className="space-y-3">
                    {tasks.map(task => (
                      <TaskCard 
                        key={task.id} 
                        task={task}
                        currentUserEmail={currentUser?.email}
                        onStatusChange={updateTaskStatus}
                        onSwapRequest={handleSwapRequest}
                        canSwap={members.length > 1}
                      />
                    ))}
                  </div>
                )}
              </CardContent>
            </Card>

            {/* Team Members */}
            <Card>
              <CardHeader>
                <CardTitle className="flex items-center gap-2">
                  <Users className="w-5 h-5 text-slate-900" />
                  Team Members
                </CardTitle>
              </CardHeader>
              <CardContent>
                <div className="space-y-3">
                  {members.map(member => {
                    const memberTasks = tasks.filter(t => t.assigned_to === member.email);
                    const completedCount = memberTasks.filter(t => t.status === 'completed').length;
                    const displayName = member.email === currentUser?.email ? 'You' : member.name;

                    return (
                      <div key={member.id} className="flex items-center gap-3 p-3 bg-slate-50 rounded-lg">
                        <div className="w-10 h-10 bg-slate-900 rounded-full flex items-center justify-center text-white font-medium">
                          {displayName.charAt(0).toUpperCase()}
                        </div>
                        <div className="flex-1 min-w-0">
                          <div className="flex items-center gap-2">
                            <p className="font-medium text-slate-900 truncate">
                              {displayName}
                              {member.is_owner && (
                                <span className="text-xs text-slate-700 ml-1">(Owner)</span>
                              )}
                            </p>
                            {member.email === currentUser?.email && (
                              <Button
                                variant="ghost"
                                size="sm"
                                onClick={() => {
                                  setMyMember(member);
                                  setShowEditMyName(true);
                                }}
                                className="h-6 px-2 text-xs"
                              >
                                <Edit className="w-3 h-3" />
                              </Button>
                            )}
                          </div>
                          <p className="text-xs text-slate-500 truncate">{member.email}</p>
                          {member.skill && (
                            <Badge variant="secondary" className="text-xs mt-1">
                              {member.skill}
                            </Badge>
                          )}
                        </div>
                        <div className="text-right text-sm">
                          <div className="font-medium text-slate-900">{completedCount}/{memberTasks.length}</div>
                          <div className="text-xs text-slate-500">tasks</div>
                        </div>
                      </div>
                    );
                  })}
                </div>
              </CardContent>
              </Card>
              </div>
              </div>
      </main>

      {/* Modals */}
      {isOwner && (
        <>
          <EditProjectModal
            open={showEditProject}
            onClose={() => setShowEditProject(false)}
            project={project}
            onSuccess={handleProjectUpdated}
          />
          <EditMembersModal
            open={showEditMembers}
            onClose={() => setShowEditMembers(false)}
            projectId={project.id}
            members={members}
            onSuccess={handleMembersUpdated}
          />
          <ManageTasksModal
            open={showManageTasks}
            onClose={() => setShowManageTasks(false)}
            projectId={project.id}
            members={members}
            onSuccess={() => {
              setShowManageTasks(false);
              loadProject();
            }}
          />
        </>
      )}

      <FairnessModal
        open={showFairnessModal}
        onClose={() => setShowFairnessModal(false)}
        project={project}
        members={members}
        tasks={tasks}
      />

      <CompletionSummaryModal
        open={showCompletionSummary}
        onClose={() => setShowCompletionSummary(false)}
        project={project}
        members={members}
        tasks={tasks}
      />

      <EditMyNameModal
        open={showEditMyName}
        onClose={() => {
          setShowEditMyName(false);
          setMyMember(null);
        }}
        member={myMember}
        onSuccess={() => {
          setShowEditMyName(false);
          setMyMember(null);
          loadProject();
        }}
      />

      {selectedTaskForSwap && (
        <TaskSwapModal
          open={showTaskSwap}
          onClose={() => {
            setShowTaskSwap(false);
            setSelectedTaskForSwap(null);
          }}
          task={selectedTaskForSwap}
          members={members}
          onSuccess={handleTaskSwapped}
        />
      )}

      {/* Delete Confirmation Dialog */}
      <AlertDialog open={showDeleteDialog} onOpenChange={setShowDeleteDialog}>
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>Delete Project?</AlertDialogTitle>
            <AlertDialogDescription>
              This will permanently delete "{project?.title}" and all its tasks. This action cannot be undone.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel>Cancel</AlertDialogCancel>
            <AlertDialogAction
              onClick={handleDeleteProject}
              disabled={isDeleting}
              className="bg-red-600 hover:bg-red-700"
            >
              {isDeleting ? (
                <>
                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                  Deleting...
                </>
              ) : (
                'Delete Project'
              )}
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>

      {/* Leave Confirmation Dialog */}
      <AlertDialog open={showLeaveDialog} onOpenChange={setShowLeaveDialog}>
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>Leave Project?</AlertDialogTitle>
            <AlertDialogDescription>
              You will be removed from "{project?.title}" and your tasks will be deleted. This action cannot be undone.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel>Cancel</AlertDialogCancel>
            <AlertDialogAction
              onClick={handleLeaveProject}
              disabled={isLeaving}
              className="bg-red-600 hover:bg-red-700"
            >
              {isLeaving ? (
                <>
                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                  Leaving...
                </>
              ) : (
                'Leave Project'
              )}
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  );
}
