import React, { useState } from 'react';
import { base44 } from '@/api/base44Client';
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';
import { Loader2, AlertCircle, CheckCircle } from 'lucide-react';
import { Alert, AlertDescription } from '@/components/ui/alert';

export default function JoinProjectModal({ open, onClose, onSuccess, userEmail, userName }) {
  const [joinCode, setJoinCode] = useState('');
  const [skill, setSkill] = useState('');
  const [isJoining, setIsJoining] = useState(false);
  const [error, setError] = useState('');
  const [success, setSuccess] = useState(false);

  const handleJoin = async () => {
    if (joinCode.length !== 4) {
      setError('Please enter a valid 4-digit code');
      return;
    }

    setIsJoining(true);
    setError('');

    try {
      // Find project with this join code
      const projects = await base44.entities.Project.filter({ join_code: joinCode });
      
      if (projects.length === 0) {
        setError('No project found with this code');
        setIsJoining(false);
        return;
      }

      const project = projects[0];

      // Check if already a member
      const existingMembers = await base44.entities.ProjectMember.filter({ 
        project_id: project.id, 
        email: userEmail 
      });

      if (existingMembers.length > 0) {
        setError('You are already a member of this project');
        setIsJoining(false);
        return;
      }

      // Add user as member
      await base44.entities.ProjectMember.create({
        project_id: project.id,
        name: userName || userEmail.split('@')[0],
        email: userEmail,
        skill: skill,
        is_owner: false,
      });

      // After joining, trigger AI to assign tasks to the new member
      try {
        const allMembers = await base44.entities.ProjectMember.filter({ project_id: project.id });
        const unassignedTasks = await base44.entities.Task.filter({ project_id: project.id });
        
        if (unassignedTasks.length > 0 && allMembers.length > 1) {
          const prompt = `You are a project management AI. A new member just joined this project. Reassign tasks fairly to balance the workload among all team members.

Project: ${project.title}
New Member: ${userName || userEmail.split('@')[0]} (${userEmail})

All Team Members:
${allMembers.map((m, i) => `${i + 1}. ${m.name} (${m.email})${m.skill ? ` - Skill: ${m.skill}` : ''}`).join('\n')}

Current Tasks:
${unassignedTasks.map((t, i) => `${i + 1}. ${t.title} - Currently assigned to: ${t.assigned_name}`).join('\n')}

Requirements:
- Reassign some tasks to the new member to balance workload
- Consider skills if available
- Make sure every member has at least one task
- Don't reassign completed tasks

Return the updated task assignments.`;

          const response = await base44.integrations.Core.InvokeLLM({
            prompt,
            response_json_schema: {
              type: 'object',
              properties: {
                assignments: {
                  type: 'array',
                  items: {
                    type: 'object',
                    properties: {
                      task_title: { type: 'string' },
                      assigned_email: { type: 'string' },
                      assigned_name: { type: 'string' }
                    }
                  }
                }
              }
            }
          });

          if (response.assignments && response.assignments.length > 0) {
            for (const assignment of response.assignments) {
              const taskToUpdate = unassignedTasks.find(t => t.title === assignment.task_title);
              if (taskToUpdate && taskToUpdate.status !== 'completed') {
                await base44.entities.Task.update(taskToUpdate.id, {
                  assigned_to: assignment.assigned_email,
                  assigned_name: assignment.assigned_name
                });
              }
            }
          }
        }
      } catch (aiError) {
        console.error('AI task assignment failed:', aiError);
        // User still joined successfully
      }

      setSuccess(true);
      setTimeout(() => {
        onSuccess(project);
        resetForm();
      }, 1500);

    } catch (err) {
      setError('Failed to join project. Please try again.');
    }
    setIsJoining(false);
  };

  const resetForm = () => {
    setJoinCode('');
    setSkill('');
    setError('');
    setSuccess(false);
  };

  const handleClose = () => {
    resetForm();
    onClose();
  };

  return (
    <Dialog open={open} onOpenChange={handleClose}>
      <DialogContent className="sm:max-w-md">
        <DialogHeader>
          <DialogTitle>Join a Project</DialogTitle>
        </DialogHeader>

        {error && (
          <Alert variant="destructive">
            <AlertCircle className="h-4 w-4" />
            <AlertDescription>{error}</AlertDescription>
          </Alert>
        )}

        {success ? (
          <div className="py-8 text-center">
            <CheckCircle className="w-16 h-16 text-emerald-500 mx-auto mb-4" />
            <h3 className="text-lg font-semibold text-slate-900">Successfully Joined!</h3>
            <p className="text-slate-600">Redirecting to project...</p>
          </div>
        ) : (
          <div className="space-y-4">
            <p className="text-sm text-slate-600">
              Enter the 4-digit code shared by your team leader.
            </p>

            <div>
              <Label htmlFor="code">Join Code</Label>
              <Input
                id="code"
                value={joinCode}
                onChange={(e) => setJoinCode(e.target.value.replace(/\D/g, '').slice(0, 4))}
                placeholder="0000"
                className="text-2xl text-center tracking-widest font-mono"
                maxLength={4}
              />
            </div>

            <div>
              <Label htmlFor="skill">Your Skill (optional)</Label>
              <Input
                id="skill"
                value={skill}
                onChange={(e) => setSkill(e.target.value)}
                placeholder="e.g., Design, Coding, Research"
              />
            </div>

            <Button
              onClick={handleJoin}
              disabled={isJoining || joinCode.length !== 4}
              className="w-full bg-slate-900 hover:bg-slate-800"
            >
              {isJoining ? (
                <>
                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                  Joining...
                </>
              ) : (
                'Join Project'
              )}
            </Button>
          </div>
        )}
      </DialogContent>
    </Dialog>
  );
}
