import React, { useState } from 'react';
import { base44 } from '@/api/base44Client';
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { Loader2, AlertCircle, ArrowLeftRight } from 'lucide-react';
import { Alert, AlertDescription } from '@/components/ui/alert';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { Label } from '@/components/ui/label';

export default function TaskSwapModal({ open, onClose, task, members, onSuccess }) {
  const [selectedMember, setSelectedMember] = useState('');
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [error, setError] = useState('');

  const handleSwap = async () => {
    if (!selectedMember) {
      setError('Please select a member to swap with');
      return;
    }

    setIsSubmitting(true);
    setError('');

    try {
      const member = members.find(m => m.email === selectedMember);
      
      await base44.entities.Task.update(task.id, {
        assigned_to: member.email,
        assigned_name: member.name,
      });

      onSuccess();
      onClose();
    } catch (err) {
      setError('Failed to swap task. Please try again.');
    }
    setIsSubmitting(false);
  };

  // Filter out current assignee
  const availableMembers = members.filter(m => m.email !== task.assigned_to);

  return (
    <Dialog open={open} onOpenChange={onClose}>
      <DialogContent className="sm:max-w-md">
        <DialogHeader>
          <DialogTitle className="flex items-center gap-2">
            <ArrowLeftRight className="w-5 h-5" />
            Swap Task Assignment
          </DialogTitle>
        </DialogHeader>

        {error && (
          <Alert variant="destructive" className="mb-4">
            <AlertCircle className="h-4 w-4" />
            <AlertDescription>{error}</AlertDescription>
          </Alert>
        )}

        <div className="space-y-4">
          <div className="p-4 bg-slate-50 rounded-lg">
            <h4 className="font-medium text-slate-900 mb-2">{task.title}</h4>
            <p className="text-sm text-slate-600">
              Currently assigned to: <span className="font-medium">{task.assigned_name}</span>
            </p>
          </div>

          <div>
            <Label>Swap with</Label>
            <Select value={selectedMember} onValueChange={setSelectedMember}>
              <SelectTrigger>
                <SelectValue placeholder="Select a team member" />
              </SelectTrigger>
              <SelectContent>
                {availableMembers.map(member => (
                  <SelectItem key={member.id} value={member.email}>
                    <div className="flex items-center gap-2">
                      <span>{member.name}</span>
                      {member.skill && (
                        <span className="text-xs text-slate-500">({member.skill})</span>
                      )}
                    </div>
                  </SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>

          <div className="flex justify-end gap-3 pt-4">
            <Button variant="outline" onClick={onClose}>
              Cancel
            </Button>
            <Button
              onClick={handleSwap}
              disabled={isSubmitting || !selectedMember}
              className="bg-slate-900 hover:bg-slate-800"
            >
              {isSubmitting ? (
                <>
                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                  Swapping...
                </>
              ) : (
                <>
                  <ArrowLeftRight className="w-4 h-4 mr-2" />
                  Swap Task
                </>
              )}
            </Button>
          </div>
        </div>
      </DialogContent>
    </Dialog>
  );
}
