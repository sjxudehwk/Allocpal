import React, { useState } from 'react';
import { Badge } from '@/components/ui/badge';
import { Button } from '@/components/ui/button';
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuTrigger,
  DropdownMenuSeparator,
} from '@/components/ui/dropdown-menu';
import { Circle, Clock, CheckCircle2, ChevronDown, User, ArrowLeftRight, FileText } from 'lucide-react';
import ProofOfWorkModal from './ProofOfWorkModal';

export default function TaskCard({ task, currentUserEmail, onStatusChange, onSwapRequest, canSwap }) {
  const isMyTask = task.assigned_to === currentUserEmail;
  const [showProofModal, setShowProofModal] = useState(false);

  const handleStatusChange = (newStatus) => {
    if (newStatus === 'completed' && !task.proof_of_work_url) {
      setShowProofModal(true);
    } else {
      onStatusChange(task.id, newStatus);
    }
  };

  const handleProofSubmitted = () => {
    setShowProofModal(false);
    onStatusChange(task.id, 'completed');
  };

  const statusConfig = {
    not_started: {
      label: 'Not Started',
      icon: Circle,
      color: 'bg-slate-100 text-slate-600',
    },
    in_progress: {
      label: 'In Progress',
      icon: Clock,
      color: 'bg-blue-100 text-blue-700',
    },
    completed: {
      label: 'Completed',
      icon: CheckCircle2,
      color: 'bg-emerald-100 text-emerald-700',
    },
  };

  const priorityConfig = {
    low: 'bg-slate-100 text-slate-600',
    medium: 'bg-amber-100 text-amber-700',
    high: 'bg-red-100 text-red-700',
  };

  const status = statusConfig[task.status] || statusConfig.not_started;
  const StatusIcon = status.icon;

  return (
    <div className={`p-4 rounded-xl border-2 transition-all ${
      isMyTask ? 'border-slate-900 bg-slate-50' : 'border-slate-100 bg-white'
    }`}>
      <div className="flex items-start justify-between gap-3 mb-3">
        <div className="flex-1">
          <h4 className="font-medium text-slate-900 mb-1">{task.title}</h4>
          {task.description && (
            <p className="text-sm text-slate-600">{task.description}</p>
          )}
        </div>
        <Badge className={priorityConfig[task.priority] || priorityConfig.medium}>
          {task.priority}
        </Badge>
      </div>

      <div className="flex items-center justify-between">
        <div className="flex items-center gap-2">
          <div className="flex items-center gap-1.5 text-sm text-slate-600">
            <User className="w-3.5 h-3.5" />
            <span>{task.assigned_name}</span>
            {isMyTask && (
              <Badge variant="secondary" className="text-xs bg-slate-900 text-white">
                You
              </Badge>
            )}
          </div>
        </div>

        {isMyTask ? (
          <div className="flex gap-2">
            {canSwap && (
              <Button
                variant="outline"
                size="sm"
                onClick={() => onSwapRequest(task)}
              >
                <ArrowLeftRight className="w-3.5 h-3.5 mr-1.5" />
                Swap
              </Button>
            )}
            <DropdownMenu>
              <DropdownMenuTrigger asChild>
                <Button 
                  variant="outline" 
                  size="sm" 
                  className={`${status.color} border-0`}
                >
                  <StatusIcon className="w-3.5 h-3.5 mr-1.5" />
                  {status.label}
                  <ChevronDown className="w-3.5 h-3.5 ml-1.5" />
                </Button>
              </DropdownMenuTrigger>
              <DropdownMenuContent align="end">
                <DropdownMenuItem onClick={() => handleStatusChange('not_started')}>
                  <Circle className="w-4 h-4 mr-2 text-slate-500" />
                  Not Started
                </DropdownMenuItem>
                <DropdownMenuItem onClick={() => handleStatusChange('in_progress')}>
                  <Clock className="w-4 h-4 mr-2 text-blue-500" />
                  In Progress
                </DropdownMenuItem>
                <DropdownMenuItem onClick={() => handleStatusChange('completed')}>
                  <CheckCircle2 className="w-4 h-4 mr-2 text-emerald-500" />
                  Completed
                </DropdownMenuItem>
              </DropdownMenuContent>
            </DropdownMenu>
          </div>
        ) : (
          <Badge className={`${status.color} border-0`}>
            <StatusIcon className="w-3.5 h-3.5 mr-1.5" />
            {status.label}
          </Badge>
        )}
      </div>

      {task.proof_of_work_url && (
        <div className="mt-3 pt-3 border-t border-slate-200">
          <div className="flex items-center justify-between">
            <a 
              href={task.proof_of_work_url} 
              target="_blank" 
              rel="noopener noreferrer"
              className="flex items-center gap-2 text-sm text-slate-600 hover:text-slate-900 transition-colors"
            >
              <FileText className="w-4 h-4" />
              View Proof of Work
            </a>
            {isMyTask && (
              <Button
                variant="ghost"
                size="sm"
                onClick={() => setShowProofModal(true)}
                className="text-xs"
              >
                Edit
              </Button>
            )}
          </div>
          {task.proof_note && (
            <p className="text-xs text-slate-500 mt-2">{task.proof_note}</p>
          )}
        </div>
      )}

      <ProofOfWorkModal
        open={showProofModal}
        onClose={() => setShowProofModal(false)}
        task={task}
        onSuccess={handleProofSubmitted}
      />
