import React, { useState, useEffect } from 'react';
import { base44 } from '@/api/base44Client';
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';
import { Textarea } from '@/components/ui/textarea';
import { Plus, Trash2, Loader2, AlertCircle, Save } from 'lucide-react';
import { Alert, AlertDescription } from '@/components/ui/alert';
import { Badge } from '@/components/ui/badge';

export default function ManageTasksModal({ open, onClose, projectId, members, onSuccess }) {
  const [tasks, setTasks] = useState([]);
  const [newTasks, setNewTasks] = useState([]);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [error, setError] = useState('');

  useEffect(() => {
    if (open) {
      loadTasks();
    }
  }, [open, projectId]);

  const loadTasks = async () => {
    try {
      const projectTasks = await base44.entities.Task.filter({ project_id: projectId });
      setTasks(projectTasks.map(t => ({ ...t, _action: 'keep' })));
      setNewTasks([]);
    } catch (err) {
      console.error('Failed to load tasks:', err);
    }
  };

  const updateTask = (index, field, value) => {
    const updated = [...tasks];
    updated[index][field] = value;
    if (updated[index]._action === 'keep') {
      updated[index]._action = 'update';
    }
    setTasks(updated);
  };

  const markForDeletion = (index) => {
    const updated = [...tasks];
    updated[index]._action = updated[index]._action === 'delete' ? 'keep' : 'delete';
    setTasks(updated);
  };

  const addNewTask = () => {
    setNewTasks([...newTasks, { title: '', description: '', priority: 'medium', assigned_to: '', assigned_name: '' }]);
  };

  const updateNewTask = (index, field, value) => {
    const updated = [...newTasks];
    updated[index][field] = value;
    
    // Auto-fill assigned_name when email is selected
    if (field === 'assigned_to') {
      const member = members.find(m => m.email === value);
      if (member) {
        updated[index].assigned_name = member.name;
      }
    }
    
    setNewTasks(updated);
  };

  const removeNewTask = (index) => {
    setNewTasks(newTasks.filter((_, i) => i !== index));
  };

  const handleSubmit = async () => {
    // Validate
    for (const task of tasks) {
      if (task._action !== 'delete' && !task.title.trim()) {
        setError('All tasks must have a title');
        return;
      }
    }
    for (const task of newTasks) {
      if (!task.title.trim() || !task.assigned_to) {
        setError('All new tasks must have title and assigned member');
        return;
      }
    }

    setIsSubmitting(true);
    setError('');

    try {
      // Update existing tasks
      for (const task of tasks) {
        if (task._action === 'update') {
          await base44.entities.Task.update(task.id, {
            title: task.title,
            description: task.description,
            priority: task.priority,
          });
        } else if (task._action === 'delete') {
          await base44.entities.Task.delete(task.id);
        }
      }

      // Add new tasks
      for (const task of newTasks) {
        await base44.entities.Task.create({
          project_id: projectId,
          title: task.title,
          description: task.description,
          priority: task.priority,
          assigned_to: task.assigned_to,
          assigned_name: task.assigned_name,
          status: 'not_started',
        });
      }

      onSuccess();
    } catch (err) {
      setError('Failed to update tasks. Please try again.');
    }
    setIsSubmitting(false);
  };

  return (
    <Dialog open={open} onOpenChange={onClose}>
      <DialogContent className="sm:max-w-3xl max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle>Manage Tasks</DialogTitle>
        </DialogHeader>

        {error && (
          <Alert variant="destructive" className="mb-4">
            <AlertCircle className="h-4 w-4" />
            <AlertDescription>{error}</AlertDescription>
          </Alert>
        )}

        <div className="space-y-6">
          {/* Existing Tasks */}
          {tasks.length > 0 && (
            <div>
              <h3 className="text-sm font-medium text-slate-700 mb-3">Current Tasks</h3>
              <div className="space-y-3 max-h-60 overflow-y-auto">
                {tasks.map((task, index) => (
                  <div 
                    key={task.id} 
                    className={`p-4 rounded-lg border-2 transition-all ${
                      task._action === 'delete' 
                        ? 'border-red-200 bg-red-50 opacity-50' 
                        : task._action === 'update'
                        ? 'border-amber-200 bg-amber-50'
                        : 'border-slate-200 bg-slate-50'
                    }`}
                  >
                    <div className="flex justify-between items-start mb-3">
                      <div className="flex items-center gap-2">
                        <Badge className="bg-slate-900 text-white text-xs">
                          {task.assigned_name}
                        </Badge>
                        {task._action === 'update' && (
                          <Badge variant="secondary" className="bg-amber-100 text-amber-700 text-xs">
                            Modified
                          </Badge>
                        )}
                      </div>
                      <Button
                        variant="ghost"
                        size="sm"
                        onClick={() => markForDeletion(index)}
                        className={task._action === 'delete' ? 'text-emerald-600' : 'text-red-500'}
                      >
                        {task._action === 'delete' ? 'Undo' : <Trash2 className="w-4 h-4" />}
                      </Button>
                    </div>
                    {task._action !== 'delete' && (
                      <div className="space-y-3">
                        <div>
                          <Label className="text-xs">Title</Label>
                          <Input
                            value={task.title}
                            onChange={(e) => updateTask(index, 'title', e.target.value)}
                          />
                        </div>
                        <div>
                          <Label className="text-xs">Description</Label>
                          <Textarea
                            value={task.description || ''}
                            onChange={(e) => updateTask(index, 'description', e.target.value)}
                            rows={2}
                          />
                        </div>
                        <div>
                          <Label className="text-xs">Priority</Label>
                          <select
                            value={task.priority}
                            onChange={(e) => updateTask(index, 'priority', e.target.value)}
                            className="w-full h-10 px-3 rounded-md border border-slate-200 bg-white text-sm"
                          >
                            <option value="low">Low</option>
                            <option value="medium">Medium</option>
                            <option value="high">High</option>
                          </select>
                        </div>
                      </div>
                    )}
                  </div>
                ))}
              </div>
            </div>
          )}

          {/* New Tasks */}
          {newTasks.length > 0 && (
            <div>
              <h3 className="text-sm font-medium text-slate-700 mb-3">New Tasks</h3>
              <div className="space-y-3 max-h-60 overflow-y-auto">
                {newTasks.map((task, index) => (
                  <div key={index} className="p-4 bg-emerald-50 border-2 border-emerald-200 rounded-lg">
                    <div className="flex justify-between items-start mb-3">
                      <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 text-xs">
                        New
                      </Badge>
                      <Button
                        variant="ghost"
                        size="sm"
                        onClick={() => removeNewTask(index)}
                        className="text-red-500 h-8 w-8 p-0"
                      >
                        <Trash2 className="w-4 h-4" />
                      </Button>
                    </div>
                    <div className="space-y-3">
                      <div>
                        <Label className="text-xs">Title *</Label>
                        <Input
                          value={task.title}
                          onChange={(e) => updateNewTask(index, 'title', e.target.value)}
                          placeholder="Task title"
                        />
                      </div>
                      <div>
                        <Label className="text-xs">Description</Label>
                        <Textarea
                          value={task.description}
                          onChange={(e) => updateNewTask(index, 'description', e.target.value)}
                          placeholder="Task details..."
                          rows={2}
                        />
                      </div>
                      <div className="grid grid-cols-2 gap-3">
                        <div>
                          <Label className="text-xs">Priority</Label>
                          <select
                            value={task.priority}
                            onChange={(e) => updateNewTask(index, 'priority', e.target.value)}
                            className="w-full h-10 px-3 rounded-md border border-slate-200 bg-white text-sm"
                          >
                            <option value="low">Low</option>
                            <option value="medium">Medium</option>
                            <option value="high">High</option>
                          </select>
                        </div>
                        <div>
                          <Label className="text-xs">Assign To *</Label>
                          <select
                            value={task.assigned_to}
                            onChange={(e) => updateNewTask(index, 'assigned_to', e.target.value)}
                            className="w-full h-10 px-3 rounded-md border border-slate-200 bg-white text-sm"
                          >
                            <option value="">Select member</option>
                            {members.map(member => (
                              <option key={member.id} value={member.email}>
                                {member.name}
                              </option>
                            ))}
                          </select>
                        </div>
                      </div>
                    </div>
                  </div>
                ))}
              </div>
            </div>
          )}

          <Button
            variant="outline"
            onClick={addNewTask}
            className="w-full"
          >
            <Plus className="w-4 h-4 mr-2" /> Add New Task
          </Button>

          <div className="flex justify-end gap-3 pt-4 border-t">
            <Button variant="outline" onClick={onClose}>
              Cancel
            </Button>
            <Button
              onClick={handleSubmit}
              disabled={isSubmitting}
              className="bg-slate-900 hover:bg-slate-800"
            >
              {isSubmitting ? (
                <>
                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                  Saving...
                </>
              ) : (
                <>
                  <Save className="w-4 h-4 mr-2" />
                  Save Changes
                </>
              )}
            </Button>
          </div>
        </div>
      </DialogContent>
    </Dialog>
  );
}
