import React, { useState, useEffect } from 'react';
import { base44 } from '@/api/base44Client';
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogDescription,
} from '@/components/ui/dialog';
import { Button } from '@/components/ui/button';
import { Label } from '@/components/ui/label';
import { Textarea } from '@/components/ui/textarea';
import { Loader2, Upload, FileText, AlertCircle } from 'lucide-react';
import { Alert, AlertDescription } from '@/components/ui/alert';

export default function ProofOfWorkModal({ open, onClose, task, onSuccess }) {
  const [file, setFile] = useState(null);
  const [note, setNote] = useState('');
  const [isUploading, setIsUploading] = useState(false);
  const [error, setError] = useState('');

  useEffect(() => {
    if (task?.proof_note) {
      setNote(task.proof_note);
    }
  }, [task]);

  const handleFileChange = (e) => {
    const selectedFile = e.target.files[0];
    if (selectedFile) {
      setFile(selectedFile);
      setError('');
    }
  };

  const handleSubmit = async () => {
    if (!file && !task.proof_of_work_url) {
      setError('Please select a file to upload');
      return;
    }

    setIsUploading(true);
    setError('');

    try {
      let file_url = task.proof_of_work_url;
      
      // Upload new file if selected
      if (file) {
        const result = await base44.integrations.Core.UploadFile({ file });
        file_url = result.file_url;
      }

      // Update task with proof of work
      await base44.entities.Task.update(task.id, {
        proof_of_work_url: file_url,
        proof_note: note,
      });

      onSuccess();
      resetForm();
    } catch (err) {
      setError('Failed to upload proof of work. Please try again.');
    }
    setIsUploading(false);
  };

  const resetForm = () => {
    setFile(null);
    setNote('');
    setError('');
  };

  const handleClose = () => {
    resetForm();
    onClose();
  };

  return (
    <Dialog open={open} onOpenChange={handleClose}>
      <DialogContent className="sm:max-w-md">
        <DialogHeader>
          <DialogTitle>{task?.proof_of_work_url ? 'Edit' : 'Submit'} Proof of Work</DialogTitle>
          <DialogDescription>
            Upload a file (screenshot, PDF, PPT, etc.) to prove you've completed this task
          </DialogDescription>
        </DialogHeader>

        {error && (
          <Alert variant="destructive" className="mb-4">
            <AlertCircle className="h-4 w-4" />
            <AlertDescription>{error}</AlertDescription>
          </Alert>
        )}

        <div className="space-y-4">
          <div>
            <Label className="mb-2 block">Task: {task?.title}</Label>
            <div className="mt-2">
              <label className="flex flex-col items-center justify-center w-full h-32 border-2 border-dashed border-slate-300 rounded-lg cursor-pointer hover:border-slate-400 transition-colors bg-slate-50">
                <div className="text-center p-4">
                  {file ? (
                    <>
                      <FileText className="w-8 h-8 text-emerald-600 mx-auto mb-2" />
                      <p className="text-sm text-slate-700 font-medium">{file.name}</p>
                      <p className="text-xs text-slate-500 mt-1">
                        {(file.size / 1024 / 1024).toFixed(2)} MB
                      </p>
                    </>
                  ) : (
                    <>
                      <Upload className="w-8 h-8 text-slate-400 mx-auto mb-2" />
                      <p className="text-sm text-slate-600">Click to upload proof</p>
                      <p className="text-xs text-slate-500 mt-1">
                        PDF, PPT, PNG, JPG, etc.
                      </p>
                    </>
                  )}
                </div>
                <input
                  type="file"
                  className="hidden"
                  onChange={handleFileChange}
                  accept="image/*,.pdf,.ppt,.pptx,.doc,.docx"
                />
              </label>
            </div>
          </div>

          {task?.proof_of_work_url && !file && (
            <div className="text-sm text-slate-600">
              Current file: <a href={task.proof_of_work_url} target="_blank" rel="noopener noreferrer" className="text-blue-600 hover:underline">View</a>
            </div>
          )}

          <div>
            <Label htmlFor="note">Note (optional)</Label>
            <Textarea
              id="note"
              value={note}
              onChange={(e) => setNote(e.target.value)}
              placeholder="Add any additional notes about your work..."
              rows={3}
            />
          </div>

          <div className="flex justify-end gap-3 pt-4">
            <Button variant="outline" onClick={handleClose}>
              Cancel
            </Button>
            <Button
              onClick={handleSubmit}
              disabled={isUploading || (!file && !task?.proof_of_work_url)}
              className="bg-slate-900 hover:bg-slate-800"
            >
              {isUploading ? (
                <>
                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                  {file ? 'Uploading...' : 'Saving...'}
                </>
              ) : (
                task?.proof_of_work_url ? 'Update Proof' : 'Submit & Complete Task'
              )}
            </Button>
          </div>
        </div>
      </DialogContent>
    </Dialog>
  );
}
